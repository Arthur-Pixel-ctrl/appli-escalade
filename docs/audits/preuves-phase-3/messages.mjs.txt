// ---------------- Comparaison ----------------
const spec = lireSpec()
const parCode = new Map()
for (const a of recoltes) {
  if (!parCode.has(a.code)) parCode.set(a.code, [])
  parCode.get(a.code).push(a)
}
const lignes = []
let ecarts = 0
for (const code of Object.keys(spec)) {
  const regle = spec[code]
  const vues = parCode.get(code) ?? []
  if (vues.length === 0) { lignes.push(`${code}\tNON DÉCLENCHÉ`); ecarts++; continue }
  const messages = [...new Set(vues.map((a) => a.message))]
  const niveaux = [...new Set(vues.map((a) => a.niveau))]
  const sources = [...new Set(vues.map((a) => a.source))]
  for (const m of messages) {
    const r = comparer(regle, m)
    if (r !== 'identique') ecarts++
    lignes.push(`${code}\t${r.toUpperCase()}\tniveau ${niveaux.join('/')} (spéc. ${regle.niveau})\t${m}`)
  }
  const niveauOk = regle.niveau === 'voir condition' || niveaux.every((x) => x === regle.niveau)
  if (!niveauOk) { ecarts++; lignes.push(`${code}\tNIVEAU DIFFÉRENT : ${niveaux} au lieu de ${regle.niveau}`) }
  lignes.push(`${code}\tsource code : ${sources.join(' / ')}\t| spéc. : ${regle.source}`)
}
const inconnus = [...parCode.keys()].filter((c) => !spec[c])
console.log(lignes.join('\n'))
console.log(`\nCodes de la spécification : ${Object.keys(spec).length} ; déclenchés : ${[...parCode.keys()].filter((c) => spec[c]).length} ; codes inconnus émis : ${inconnus.join(', ') || 'aucun'} ; écarts : ${ecarts}`)
