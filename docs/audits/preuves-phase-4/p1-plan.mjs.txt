// Parcours 1 (ordinateur, 1280 × 800) : premier lancement sans douleur, profil d'un grimpeur confirmé,
// plan proposé (doigts, puissance, résistance ; 3 séances par semaine) puis validé.
// L'état de l'appareil (IndexedDB compris) est gardé dans etat-p1.json pour les parcours suivants.
// Lancer : node /home/claude/audits/phase-4/p1-plan.mjs (serveur : npx vite preview --port 4193)
import { ouvrir, etapes, texte, lireTable, URL_APP, DOSSIER, petitsBoutons } from './lib.mjs'

const { navigateur, contexte, page: p, erreurs } = await ouvrir()
const { etape, bilan } = etapes(p, 'p1')
await p.clock.setFixedTime(new Date('2026-12-01T09:00:00'))
await p.goto(URL_APP)

await etape('premier lancement : profil confirmé, aucune douleur', async () => {
  await p.getByLabel('Prénom').fill('Arthur')
  await p.getByLabel('Niveau max en bloc').selectOption('7A')
  for (let i = 0; i < 6; i++) await p.getByRole('button', { name: 'Une année de plus' }).click()
  await p.getByRole('button', { name: 'Souvent' }).click()
  for (let i = 0; i < 15; i++) await p.getByRole('button', { name: 'Une traction de plus' }).click()
  for (const legende of ['Tu tiens 15 s sur une réglette de 25 mm, poids du corps ?', 'Tu as 18 ans ou plus ?', 'Tu sais grimper et chuter en tête ?']) {
    await p.getByRole('group', { name: legende }).getByRole('button', { name: 'Oui' }).click()
  }
  await p.getByRole('button', { name: 'Continuer' }).click()
  await p.getByRole('button', { name: 'J’ai compris' }).click()
})
await etape('plan proposé (« Proposer un plan »), niveau détaillé par défaut', async () => {
  await p.getByRole('navigation').getByRole('button', { name: 'Plan' }).first().click()
  await p.getByRole('button', { name: 'Crée ton premier cycle' }).click()
  await p.getByLabel('Nom du cycle').fill('Hiver')
  await p.getByLabel('Phase 1').selectOption({ label: 'Doigts' }).catch(async () => console.log('      options :', await p.getByLabel('Phase 1').locator('option').allInnerTexts()))
  await p.getByLabel('Phase 2').selectOption({ label: 'Puissance' }).catch(() => {})
  await p.getByLabel('Phase 3').selectOption({ label: 'Résistance' }).catch(() => {})
  console.log('      début proposé :', await p.getByLabel('Début (un lundi)').inputValue())
  await p.getByRole('button', { name: 'Proposer un plan' }).click()
  await p.getByText(/^Plan proposé/).first().waitFor()
})
await etape('valider le cycle', async () => {
  await p.getByRole('button', { name: '12 semaines' }).click().catch(() => {})
  await p.getByRole('button', { name: 'Valider le cycle' }).click()
  await p.waitForTimeout(400)
  const c = p.getByRole('dialog').getByRole('button', { name: /Valider/ })
  if (await c.count()) await c.first().click()
  await p.waitForTimeout(600)
  const cycles = await lireTable(p, 'cycles')
  console.log('      cycles :', JSON.stringify(cycles.map((c) => ({ nom: c.nom, debut: c.debut, etat: c.etat, niveau: c.niveauDetail }))))
})
await etape('boutons de moins de 48 px (ordinateur, Plan)', async () => { console.log('      ', JSON.stringify(await petitsBoutons(p))) })

const seances = await lireTable(p, 'seancesPlanifiees')
const sem1 = seances.filter((s) => s.date < '2026-12-21').sort((a, b) => a.date.localeCompare(b.date))
for (const s of sem1) {
  const refs = s.solo.etapes.map((e) => e.origine?.refId).filter(Boolean)
  console.log(`   ${s.date} ${s.solo.nom} [${s.solo.durete}] st=${s.seanceTypeId ?? '-'} groupe=${s.groupe ? s.groupe.nom + (s.groupe.miniBloc ? ' miniBloc ' + s.groupe.miniBloc.moment + ' ' + s.groupe.miniBloc.etapes.map((e) => e.origine?.refId).join(',') : '') + ' obj ' + s.groupe.objectifs.map((o) => o.ficheReference).join(',') : 'non'} solo=${refs.join(',')}`)
}
await contexte.storageState({ path: DOSSIER + 'etat-p1.json', indexedDB: true })
console.log('erreurs de page :', erreurs.length ? erreurs : 'aucune')
console.log('bilan :', bilan.ok, 'ok,', bilan.echecs.length, 'échec(s)', bilan.echecs)
await navigateur.close()
