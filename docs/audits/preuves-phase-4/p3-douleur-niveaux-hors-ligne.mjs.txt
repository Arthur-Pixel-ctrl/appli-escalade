// Focus visible : tabulation sur Aujourd'hui
await etape('focus visible au clavier sur Aujourd’hui (10 tabulations)', async () => {
  const styles = []
  await p.locator('body').click({ position: { x: 5, y: 5 } })
  for (let i = 0; i < 10; i++) {
    await p.keyboard.press('Tab')
    styles.push(await p.evaluate(() => { const e = document.activeElement; if (e === document.body) return '(hors de la page):-'; const s = getComputedStyle(e); return `${(e.textContent || e.getAttribute('aria-label') || '').trim().slice(0, 18)}:${s.outlineStyle !== 'none' && s.outlineWidth !== '0px' ? 'contour ' + s.outlineWidth : s.boxShadow !== 'none' ? 'ombre' : 'RIEN'}` }))
  }
  console.log('      ', styles.join(' | '))
  if (styles.some((s) => s.endsWith('RIEN'))) throw new Error('focus sans repère')
})
