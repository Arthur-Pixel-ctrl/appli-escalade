// Parcours 2 (téléphone, 390 × 844), semaine 1 du cycle, sans compte (comme Arthur aujourd'hui) :
// séance du jour, forme du jour et J-01, séance en groupe avec mini-bloc avant (ST-DOI-02) puis débrief,
// jour de repos, déplacer et sauter depuis Programme, mini-bloc après (ST-TRA-02), séance libre.
// Lancer : node /home/claude/audits/phase-4/p2-semaine.mjs (après p1-plan.mjs)
import { ouvrir, etapes, texte, lireTable, URL_APP, DOSSIER, TELEPHONE, petitsBoutons } from './lib.mjs'

const { navigateur, contexte, page: p, erreurs } = await ouvrir({ etat: DOSSIER + 'etat-p1.json', viewport: TELEPHONE })
const { etape, bilan } = etapes(p, 'p2')
const capture = (nom) => p.screenshot({ path: `${DOSSIER}p2-${nom}.png`, fullPage: true })
const aller = async (jour) => { await p.clock.setFixedTime(new Date(`${jour}T09:00:00`)); await p.goto(URL_APP); await p.getByRole('navigation').waitFor(); await p.waitForTimeout(900) }
const onglet = (nom) => p.getByRole('navigation').getByRole('button', { name: nom }).first().click()
const petits = []
const contient = (t, a) => t.toLowerCase().includes(a.toLowerCase())
// Trois questions : effort, peau, douleur « Non, rien », puis enregistrer.
const troisQuestions = async (effort, douleur = 'Non, rien') => {
  await p.getByRole('group', { name: 'La séance était dure comment ?' }).getByRole('button', { name: String(effort), exact: true }).click()
  await p.getByRole('group', { name: 'État de la peau' }).getByRole('button', { name: 'Bonne' }).click()
  if (douleur) await p.getByRole('button', { name: douleur }).click()
  await p.getByRole('button', { name: /Enregistrer et voir le débrief/ }).click()
}
const noterPetits = async (ecran) => { const l = await petitsBoutons(p); if (l.length) petits.push({ ecran, l }) }

// ——— Mardi 8 décembre : ST-DOI-02, séance dure ———
await aller('2026-12-08')
await etape('Aujourd’hui : séance du jour, mention de phase, ligne Forme du jour, deux modes', async () => {
  const t = await texte(p)
  console.log('      ', t.slice(0, 420))
  for (const attendu of ['Phase 1 · doigts · semaine 1 sur 4', 'Suspensions max', 'Forme du jour', 'À renseigner', 'Je grimpe en groupe', 'Je grimpe seul', 'Cette semaine']) {
    if (!contient(t, attendu)) throw new Error('absent : ' + attendu)
  }
  if (/retir/i.test(t)) throw new Error('un texte parle de retirer : ' + t)
  await capture('1-aujourdhui'); await noterPetits('Aujourd’hui')
})
await etape('Forme du jour : 5 questions, « 30 secondes », Enregistrer désactivé tant qu’il manque une note', async () => {
  await p.getByRole('button', { name: /Forme du jour/ }).click()
  await p.getByRole('heading', { name: 'Forme du jour' }).waitFor()
  const focus = await p.evaluate(() => document.activeElement?.textContent)
  console.log('      focus à l’ouverture :', focus)
  const t = await texte(p)
  for (const q of ['30 secondes', 'Sommeil cette nuit', 'Fatigue', 'Courbatures', 'Douleur', 'Peau des doigts', 'Heures de sommeil (facultatif)', 'Réponds aux cinq questions pour enregistrer.']) if (!contient(t, q)) throw new Error('absent : ' + q)
  if (!(await p.getByRole('button', { name: 'Enregistrer' }).isDisabled())) throw new Error('Enregistrer actif sans réponse')
  await noterPetits('Forme du jour')
})
await etape('Forme du jour : heures hors bornes refusées, puis sommeil 2, fatigue 2 → « 2 voyants au rouge » et J-01 en information', async () => {
  const note = (q, n) => p.getByRole('group', { name: q }).getByRole('button', { name: String(n), exact: true }).click()
  await note('Sommeil cette nuit', 2); await note('Fatigue', 2); await note('Courbatures', 4); await note('Douleur', 4); await note('Peau des doigts', 4)
  await p.getByLabel('Heures de sommeil (facultatif)').fill('15')
  await p.getByRole('button', { name: 'Enregistrer' }).click()
  await p.getByText('Entre 0 et 14 h.').waitFor()
  await p.getByLabel('Heures de sommeil (facultatif)').fill('5,5')
  await p.getByRole('button', { name: 'Enregistrer' }).click()
  await p.getByText('2 voyants au rouge').waitFor()
  await p.getByText('Deux voyants sont au rouge ce matin.').waitFor()
  const formes = await lireTable(p, 'formes')
  console.log('      base :', JSON.stringify(formes.map((f) => ({ id: f.id, date: f.date, sommeil: f.sommeil, heures: f.heuresSommeil, aEnvoyer: f.aEnvoyer }))))
  const desactives = await p.locator('main button:disabled').count()
  console.log('      boutons désactivés sur Aujourd’hui :', desactives)
  await capture('2-j01')
})
await etape('Groupe (mode habituel) : mini-bloc d’abord, « Stop » en haut, « Passer au groupe »', async () => {
  await p.getByRole('button', { name: /Je grimpe en groupe/ }).click()
  await p.getByRole('heading', { name: /· avant le groupe/ }).waitFor()
  const t = await texte(p)
  console.log('      ', t.slice(0, 300))
  const stop = await p.getByRole('button', { name: 'Stop, ça fait mal' }).boundingBox()
  console.log('      Stop : y =', Math.round(stop.y), 'h =', Math.round(stop.height), '; rappel bloc sur le mini-bloc :', t.includes('Rappel de sécurité') || t.includes('sécurité du bloc'))
  await capture('3-mini-bloc'); await noterPetits('Mini-bloc')
  await p.getByRole('button', { name: 'C’est fait' }).first().click()
  await p.getByRole('button', { name: 'Passer au groupe' }).click()
})
await etape('Groupe : rappel de sécurité du bloc en tête, objectifs, « Stop » visible', async () => {
  await p.getByRole('button', { name: 'Stop, ça fait mal' }).waitFor()
  const t = await texte(p)
  console.log('      ', t.slice(0, 500))
  if (!/sécurité/i.test(t)) throw new Error('pas de rappel de sécurité du bloc')
  await capture('4-groupe'); await noterPetits('Groupe')
})
await etape('Groupe : un bloc noté, Terminer, trois questions, débrief (étapes du mini-bloc, séance prévue, mention d’envoi)', async () => {
  const bouton = p.getByRole('button', { name: /Enchaîné|Flashé/ }).first()
  if (await bouton.count()) await bouton.click()
  await p.getByRole('button', { name: /Terminer la séance/ }).click()
  const conf = p.getByRole('dialog').getByRole('button', { name: /Terminer/ })
  if (await conf.count()) await conf.first().click()
  await troisQuestions(7)
  await p.getByRole('heading', { name: 'Débrief' }).waitFor()
  const t = await texte(p)
  console.log('      ', t.slice(0, 500))
  for (const a of ['Séance prévue : mardi', '· faite', 'Enregistrée sur ce téléphone']) if (!contient(t, a)) throw new Error('absent : ' + a)
  if (contient(t, 'envoi au retour du réseau') || contient(t, '· envoyée')) throw new Error('mention d’envoi alors qu’aucun compte')
  const r = (await lireTable(p, 'seancesRealisees'))[0]
  console.log('      réalisée :', JSON.stringify({ mode: r.mode, seancePlanifieeId: !!r.seancePlanifieeId, qualite: r.qualite, marqueurs: r.marqueurs, etapes: r.etapes.map((e) => e.titre + ':' + e.statut), blocs: r.blocs.length }))
  await capture('5-debrief'); await noterPetits('Débrief')
  await p.getByRole('button', { name: 'Terminer' }).click()
  await p.getByText('Séance faite').waitFor()
})

// ——— Mercredi 9 : repos ———
await aller('2026-12-09')
await etape('Repos : « Repos », ligne Forme du jour présente, « Voir la semaine »', async () => {
  const t = await texte(p)
  for (const a of ['Repos', 'Forme du jour', 'À renseigner', 'Voir la semaine', 'Faire une séance quand même']) if (!contient(t, a)) throw new Error('absent : ' + a)
  await capture('6-repos')
})
await etape('Programme : aperçu de jeudi 10 (à venir) → avertissements, Déplacer, Sauter ; jours proposés', async () => {
  await onglet('Programme')
  await p.getByRole('button', { name: /Bloc limite \+ tractions lestées/ }).first().click()
  const t = await texte(p)
  console.log('      ', t.slice(0, 600))
  await p.getByRole('button', { name: 'Déplacer' }).click()
  const jours = await p.getByRole('group', { name: /Déplacer cette séance/ }).getByRole('button').allInnerTexts()
  console.log('      jours proposés :', jours.join(' | '))
  await capture('7-deplacer'); await noterPetits('Programme, déplacer')
  await p.getByRole('button', { name: 'vendredi 11' }).click()
  await p.getByText('Séance déplacée à vendredi 11.').waitFor()
  const t2 = await texte(p)
  if (t2.includes('visible sur ton ordinateur')) throw new Error('phrase de l’ordinateur sans compte')
  const s = (await lireTable(p, 'seancesPlanifiees')).find((x) => x.date === '2026-12-11')
  console.log('      base : date', s?.date, 'aEnvoyer', s?.aEnvoyer, 'semaine', s?.semaine)
})
await etape('Programme : samedi 12 → Sauter (confirmation) → « Séance sautée. » et « Annuler »', async () => {
  await p.getByRole('button', { name: '← Retour à la semaine' }).click()
  await p.getByRole('button', { name: /1 min \/ 1 min/ }).first().click()
  await p.getByRole('button', { name: 'Sauter cette séance' }).click()
  const d = p.getByRole('dialog')
  console.log('      fenêtre :', (await d.innerText()).replace(/\n+/g, ' / '))
  await d.getByRole('button', { name: 'Sauter la séance' }).click()
  await p.getByText('Séance sautée.').first().waitFor()
  await p.getByRole('button', { name: 'Annuler le saut de cette séance' }).waitFor()
  await capture('8-sautee')
})
await etape('Aujourd’hui : la semaine montre vendredi (déplacée) et samedi sauté', async () => {
  await onglet('Aujourd’hui')
  await p.getByText('Cette semaine').waitFor()
  const cases = await p.getByRole('list', { name: 'Séances de la semaine' }).getByRole('listitem').evaluateAll((l) => l.map((x) => x.getAttribute('aria-label')))
  console.log('      ', cases.join(' | '))
  if (!cases.some((c) => c.startsWith('Vendredi, Bloc limite'))) throw new Error('vendredi absent')
  if (!cases.some((c) => c.startsWith('Samedi') && c.endsWith('sautée'))) throw new Error('samedi pas sauté')
})
await etape('Séance libre un jour de repos (régression) : groupe libre, Terminer, questions, débrief', async () => {
  await p.getByRole('button', { name: 'Faire une séance quand même' }).click()
  await p.getByRole('button', { name: 'Stop, ça fait mal' }).waitFor()
  const t = await texte(p)
  if (/sécurité du bloc/i.test(t)) throw new Error('rappel de sécurité sur une séance libre')
  await p.getByRole('button', { name: /Terminer la séance/ }).click()
  const conf = p.getByRole('dialog').getByRole('button', { name: /Terminer/ })
  if (await conf.count()) await conf.first().click()
  await troisQuestions(4)
  await p.getByRole('heading', { name: 'Débrief' }).waitFor()
  const t2 = await texte(p)
  if (t2.includes('Séance prévue')) throw new Error('séance libre rattachée au plan')
  await p.getByRole('button', { name: 'Terminer' }).click()
})

// ——— Vendredi 11 : ST-TRA-02 déplacée, en groupe : mini-bloc APRÈS ———
await aller('2026-12-11')
await etape('Vendredi : la séance déplacée est celle du jour ; groupe d’abord, puis mini-bloc après', async () => {
  const t = await texte(p)
  if (!contient(t, 'Bloc limite + tractions lestées')) throw new Error('séance déplacée absente : ' + t.slice(0, 200))
  await p.getByRole('button', { name: /Je grimpe en groupe/ }).click()
  await p.getByRole('button', { name: 'Stop, ça fait mal' }).waitFor()
  const tg = await texte(p)
  console.log('      groupe :', tg.slice(0, 200))
  if (!/sécurité/i.test(tg)) throw new Error('pas de rappel sur le groupe')
  await p.getByRole('button', { name: /Terminer la séance/ }).click()
  const conf = p.getByRole('dialog').getByRole('button', { name: /Terminer/ })
  if (await conf.count()) await conf.first().click()
  await p.getByRole('heading', { name: /· après le groupe/ }).waitFor()
  console.log('      mini-bloc après :', (await texte(p)).slice(0, 200))
  await p.getByRole('button', { name: 'Terminer la séance' }).click()
  await troisQuestions(8)
  await p.getByRole('heading', { name: 'Débrief' }).waitFor()
  const r = (await lireTable(p, 'seancesRealisees')).find((x) => x.date === '2026-12-11' || x.creeLe?.startsWith('2026-12-11'))
  console.log('      réalisée :', JSON.stringify({ mode: r?.mode, etapes: r?.etapes.map((e) => e.titre + ':' + e.statut), marqueurs: r?.marqueurs }))
  await p.getByRole('button', { name: 'Terminer' }).click()
})

// ——— Lundi 14 : export de la sauvegarde du téléphone (pour l'ordinateur, sans compte) ———
await aller('2026-12-14')
await etape('Profil : « Sauvegarder mes données » produit un fichier', async () => {
  await onglet('Profil')
  const [dl] = await Promise.all([p.waitForEvent('download'), p.getByRole('button', { name: /Sauvegarder mes données/ }).click()])
  await dl.saveAs(DOSSIER + 'sauvegarde-telephone.json')
  console.log('      fichier :', dl.suggestedFilename())
})

console.log('boutons de moins de 48 px :', petits.length ? JSON.stringify(petits) : 'aucun')
await contexte.storageState({ path: DOSSIER + 'etat-p2.json', indexedDB: true })
console.log('erreurs de page :', erreurs.length ? erreurs : 'aucune')
console.log('bilan :', bilan.ok, 'ok,', bilan.echecs.length, 'échec(s)', bilan.echecs)
await navigateur.close()
